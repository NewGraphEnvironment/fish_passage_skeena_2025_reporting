Appendix - Fish Species by Watershed Group

Historical fish observations recorded in the watershed groups covered by this project (MoE 2024a). Each watershed group has its own column.


fiss_species_table <- readr::read_csv('data/inputs_extracted/fiss_species_table.csv',
                                      show_col_types = FALSE)

# Build the caption from the watershed-group columns actually present in the
# cached extract, NOT from params/wsg_names. If the extract is refreshed with
# different coverage the caption follows it, so the table cannot claim
# watershed groups the data does not contain.
cols_meta <- c("Scientific Name", "Species Name", "BC List", "COSEWIC")
wsg_in_table <- setdiff(names(fiss_species_table), cols_meta)

my_caption <- paste0(
  "Fish species recorded in the ",
  knitr::combine_words(wsg_in_table),
  " watershed groups."
)

my_footnote <- 'COSEWIC abbreviations :
                 SC - Special concern
                 DD - Data deficient
                 NAR - Not at risk
                 E - Endangered
                 T - Threatened

                 BC List definitions :
                 Yellow - Species that is apparently secure
                 Blue - Species that is of special concern
                 Exotic - Species that have been moved beyond their natural range as a result of human activity
                 '

# One column per watershed group is too wide for paged.js, so the PDF branch
# collapses presence into a single comma-separated column. One wide source of
# truth, pivoted only where it has to be.
tab_fiss_species <- if (isTRUE(gitbook_on)) {
  fiss_species_table
} else {
  fiss_species_table |>
    tidyr::pivot_longer(dplyr::all_of(wsg_in_table),
                        names_to = "wsg", values_to = "present") |>
    dplyr::filter(!is.na(present), trimws(as.character(present)) != "", present != "-") |>
    dplyr::group_by(dplyr::across(dplyr::all_of(cols_meta))) |>
    dplyr::summarise(`Present in WSGs` = paste(sort(unique(wsg)), collapse = ", "),
                     .groups = "drop")
}

tab_fiss_species |>
  fpr::fpr_kable(caption_text = my_caption,
                 footnote_text = my_footnote,
                 scroll = gitbook_on)
